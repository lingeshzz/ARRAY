public class Demo {
    public static void main(String[] args) {

        int a = 121;
        int n = a;
        int r = 0;

        while (a > 0) {
            int rem = a % 10;
            r = r * 10 + rem;
            a = a / 10;
        }

        if (n == r) {
            System.out.println(true);
        } else {
            System.out.println(false);
        }
    }
}
